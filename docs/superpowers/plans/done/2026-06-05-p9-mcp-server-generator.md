# P9 — MCP Server Generator (OAS Collection → Runnable MCP Server) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
> **DoD standard:** see `docs/superpowers/DEFINITION-OF-DONE.md`. A step is `[x]` only when every DoD sub-item beneath it is also `[x]`.

**Goal:** Given a collection (optionally with an `_openapiSpec`) or a bare list of requests, generate a self-contained TypeScript/Node.js MCP Server that exposes each endpoint as an AI-invocable tool — ready to be used with Cursor, Claude Code, Windsurf, or any MCP-compatible AI agent.

**Architecture:** A new `MCPServerGenerator` Go struct provides a `Generate(inputJSON, outputDir string) error` Wails binding. It uses Go `text/template` to emit a Node.js project: `index.ts`, `package.json`, `.env.example`, `README.md`. The generator lives in `internal/mcp/generator/`. A new `McpServerGenPanel.tsx` component inside the MCP panel provides the UI (collection picker, endpoint checkboxes, output directory picker, Generate button). The existing `App.OpenDirectoryDialog` Wails binding handles directory selection.

**Tech Stack:** Go `text/template`, TypeScript, React. Generated output uses `@modelcontextprotocol/sdk` (Node.js). No new Go dependencies required.

  ```
  import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
  import {
    CallToolRequestSchema,
    ListToolsRequestSchema,
  } from "@modelcontextprotocol/sdk/types.js";
  import * as dotenv from "dotenv";

  dotenv.config();

  const server = new Server(
    { name: "{{.ServerName}}", version: "1.0.0" },
    { capabilities: { tools: {} } }
  );

  server.setRequestHandler(ListToolsRequestSchema, async () => ({
    tools: [
  {{- range .Tools}}
      {
        name: "{{.Name}}",
        description: "{{.Description}}",
        inputSchema: {
          type: "object",
          properties: {{.InputSchemaJSON}},
          required: {{.RequiredJSON}},
        },
      },
  {{- end}}
    ],
  }));

  server.setRequestHandler(CallToolRequestSchema, async (request) => {
    const { name, arguments: args } = request.params;
    const a = (args ?? {}) as Record<string, string>;

  {{- range .Tools}}
    if (name === "{{.Name}}") {
      const url = `{{.URLTemplate}}`;
      const headers: Record<string, string> = { "Content-Type": "application/json" {{- if .AuthHeader}}, {{.AuthHeader}}{{- end}} };
  {{- if .HasBody}}
      const body = JSON.stringify({{.BodyTemplate}});
      const resp = await fetch(url, { method: "{{.Method}}", headers, body });
  {{- else}}
      const resp = await fetch(url, { method: "{{.Method}}", headers });
  {{- end}}
      const text = await resp.text();
      return { content: [{ type: "text", text }] };
    }
  {{- end}}

    throw new Error(`Unknown tool: ${name}`);
  });

  const transport = new StdioServerTransport();
  await server.connect(transport);
  ```

  ```go
  package generator

  import (
      "bytes"
      "encoding/json"
      "fmt"
      "os"
      "path/filepath"
      "regexp"
      "strings"
      "text/template"
  )

  // ToolDef describes a single MCP tool derived from a collection request.
  type ToolDef struct {
      Name           string
      Description    string
      Method         string
      OriginalURL    string
      URLTemplate    string // JS template literal with path params substituted
      InputSchemaJSON string
      RequiredJSON   string
      HasBody        bool
      BodyTemplate   string
      AuthHeader     string // e.g. `"Authorization": \`Bearer ${process.env.BEARER_TOKEN}\``
  }

  // EnvVar describes an env variable needed by the generated server.
  type EnvVar struct {
      Name         string
      DefaultValue string
      Description  string
  }

  // ServerSpec is the data passed to all templates.
  type ServerSpec struct {
      ServerName string
      Tools      []ToolDef
      EnvVars    []EnvVar
  }

  var slugRe = regexp.MustCompile(`[^a-zA-Z0-9_]+`)

  func slugify(s string) string {
      s = strings.TrimSpace(s)
      s = slugRe.ReplaceAllString(s, "_")
      s = strings.Trim(s, "_")
      if s == "" {
          return "tool"
      }
      return strings.ToLower(s)
  }

  type rawRequest struct {
      Name        string            `json:"name"`
      Description string            `json:"description"`
      Method      string            `json:"method"`
      URL         string            `json:"url"`
      Headers     []rawHeader       `json:"headers"`
      Auth        rawAuth           `json:"auth"`
      Body        rawBody           `json:"body"`
  }

  type rawHeader struct {
      Key   string `json:"key"`
      Value string `json:"value"`
  }

  type rawAuth struct {
      Type  string `json:"type"`  // bearer, apikey, basic, none
      Token string `json:"token"`
      Key   string `json:"key"`
      Value string `json:"value"`
  }

  type rawBody struct {
      Mode string `json:"mode"` // raw, form
      Raw  string `json:"raw"`
  }

  type GenerateInput struct {
      ServerName string       `json:"serverName"`
      Requests   []rawRequest `json:"requests"`
  }

  // Generate reads inputJSON (GenerateInput), renders templates, and writes the
  // generated server to outputDir. Creates outputDir if it does not exist.
  func Generate(inputJSON, outputDir string) error {
      var input GenerateInput
      if err := json.Unmarshal([]byte(inputJSON), &input); err != nil {
          return fmt.Errorf("invalid input JSON: %w", err)
      }
      if input.ServerName == "" {
          input.ServerName = "my-api-server"
      }

      spec, err := buildSpec(input)
      if err != nil {
          return err
      }

      if err := os.MkdirAll(outputDir, 0755); err != nil {
          return fmt.Errorf("create output dir: %w", err)
      }

      // Determine templates directory relative to this file at build time.
      // Embedded via go:embed.
      return renderAll(spec, outputDir)
  }

  func buildSpec(input GenerateInput) (ServerSpec, error) {
      spec := ServerSpec{ServerName: slugify(input.ServerName)}
      seen := map[string]int{}
      envVarSet := map[string]bool{}

      for _, req := range input.Requests {
          name := slugify(req.Name)
          if name == "" {
              name = slugify(req.Method + "_" + req.URL)
          }
          if seen[name] > 0 {
              seen[name]++
              name = fmt.Sprintf("%s_%d", name, seen[name])
          } else {
              seen[name] = 1
          }

          desc := req.Description
          if desc == "" {
              desc = fmt.Sprintf("%s %s", strings.ToUpper(req.Method), req.URL)
          }

          props := map[string]any{}
          required := []string{}

          // Extract path params from URL
          pathRe := regexp.MustCompile(`\{(\w+)\}|:(\w+)`)
          for _, m := range pathRe.FindAllStringSubmatch(req.URL, -1) {
              pname := m[1]
              if pname == "" { pname = m[2] }
              props[pname] = map[string]string{"type": "string", "description": "Path parameter"}
              required = append(required, pname)
          }

          // Convert path params to JS template literal substitutions
          urlTpl := pathRe.ReplaceAllStringFunc(req.URL, func(s string) string {
              inner := strings.Trim(strings.Trim(s, "{}"), ":")
              return "${a." + inner + "}"
          })

          // Body schema from raw JSON
          hasBody := false
          bodyTpl := "{}"
          if req.Body.Mode == "raw" && strings.TrimSpace(req.Body.Raw) != "" {
              hasBody = true
              var bodyMap map[string]any
              if err := json.Unmarshal([]byte(req.Body.Raw), &bodyMap); err == nil {
                  for k := range bodyMap {
                      props[k] = map[string]string{"type": "string", "description": "Body field"}
                  }
                  // Build JS template: { key: a.key, ... }
                  var parts []string
                  for k := range bodyMap {
                      parts = append(parts, fmt.Sprintf("%q: a.%s", k, k))
                  }
                  bodyTpl = "{ " + strings.Join(parts, ", ") + " }"
              }
          }

          propsJSON, _ := json.Marshal(props)
          reqJSON, _ := json.Marshal(required)

          authHeader := ""
          switch req.Auth.Type {
          case "bearer":
              if !envVarSet["BEARER_TOKEN"] {
                  spec.EnvVars = append(spec.EnvVars, EnvVar{Name: "BEARER_TOKEN", DefaultValue: "", Description: "Bearer token for authentication"})
                  envVarSet["BEARER_TOKEN"] = true
              }
              authHeader = `"Authorization": ` + "`Bearer ${process.env.BEARER_TOKEN}`"
          case "apikey":
              envKey := "API_KEY"
              if !envVarSet[envKey] {
                  spec.EnvVars = append(spec.EnvVars, EnvVar{Name: envKey, DefaultValue: "", Description: "API key for authentication"})
                  envVarSet[envKey] = true
              }
              authHeader = fmt.Sprintf(`%q: `, req.Auth.Key) + "`${process.env.API_KEY}`"
          }

          spec.Tools = append(spec.Tools, ToolDef{
              Name:            name,
              Description:     desc,
              Method:          strings.ToUpper(req.Method),
              OriginalURL:     req.URL,
              URLTemplate:     fmt.Sprintf("`%s`", urlTpl),
              InputSchemaJSON: string(propsJSON),
              RequiredJSON:    string(reqJSON),
              HasBody:         hasBody,
              BodyTemplate:    bodyTpl,
              AuthHeader:      authHeader,
          })
      }

      return spec, nil
  }
  ```

  ```go
  func renderAll(spec ServerSpec, outputDir string) error {
      type fileSpec struct {
          tmplName   string
          outputName string
      }
      files := []fileSpec{
          {"templates/index.ts.tmpl", "index.ts"},
          {"templates/package.json.tmpl", "package.json"},
          {"templates/env.example.tmpl", ".env.example"},
          {"templates/README.md.tmpl", "README.md"},
      }

      funcMap := template.FuncMap{
          "len": func(v interface{}) int {
              if s, ok := v.([]ToolDef); ok { return len(s) }
              return 0
          },
      }

      for _, f := range files {
          raw, err := templateFS.ReadFile(f.tmplName)
          if err != nil {
              return fmt.Errorf("read template %s: %w", f.tmplName, err)
          }
          t, err := template.New(f.tmplName).Funcs(funcMap).Delims("{{", "}}").Parse(string(raw))
          if err != nil {
              return fmt.Errorf("parse template %s: %w", f.tmplName, err)
          }
          var buf bytes.Buffer
          if err := t.Execute(&buf, spec); err != nil {
              return fmt.Errorf("execute template %s: %w", f.tmplName, err)
          }
          outPath := filepath.Join(outputDir, f.outputName)
          if err := os.WriteFile(outPath, buf.Bytes(), 0644); err != nil {
              return fmt.Errorf("write %s: %w", outPath, err)
          }
      }
      return nil
  }
  ```

  ```tsx
  import { useState } from 'react'
  import { FolderOpen, Wand2, CheckCircle2, AlertCircle } from 'lucide-react'
  import * as MCPServerGenerator from '@/wailsjs/go/main/MCPServerGenerator'
  import * as AppBinding from '@/wailsjs/go/main/App'
  import { useCollectionsStore } from '@/stores/collections'
  import { cn } from '@/lib/utils'

  interface RequestItem {
    id: string
    name: string
    method?: string
    url?: string
    description?: string
    auth?: { type: string; token?: string; key?: string; value?: string }
    body?: { mode: string; raw: string }
  }

  function flattenRequests(children: unknown[]): RequestItem[] {
    const result: RequestItem[] = []
    for (const child of children) {
      const c = child as { type?: string; id?: string; name?: string; method?: string; url?: string; description?: string; auth?: unknown; body?: unknown; children?: unknown[] }
      if (c.type === 'request' || !c.children) {
        result.push({ id: c.id ?? '', name: c.name ?? '', method: c.method, url: c.url, description: c.description, auth: c.auth as RequestItem['auth'], body: c.body as RequestItem['body'] })
      } else if (c.children) {
        result.push(...flattenRequests(c.children))
      }
    }
    return result
  }

  export function McpServerGenPanel() {
    const collections = useCollectionsStore((s) => s.collections)

    const [selectedCollectionId, setSelectedCollectionId] = useState('')
    const [serverName, setServerName] = useState('my-api-server')
    const [outputDir, setOutputDir] = useState('')
    const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
    const [generating, setGenerating] = useState(false)
    const [result, setResult] = useState<{ success: boolean; message: string } | null>(null)

    const selectedCollection = collections.find((c) => c.id === selectedCollectionId)
    const requests = selectedCollection ? flattenRequests(selectedCollection.children ?? []) : []

    const toggleAll = () => {
      if (selectedIds.size === requests.length) {
        setSelectedIds(new Set())
      } else {
        setSelectedIds(new Set(requests.map((r) => r.id)))
      }
    }

    const toggleOne = (id: string) => {
      setSelectedIds((prev) => {
        const next = new Set(prev)
        next.has(id) ? next.delete(id) : next.add(id)
        return next
      })
    }

    const handlePickDir = async () => {
      try {
        const dir = await (AppBinding as { OpenDirectoryDialog?: (title: string) => Promise<string> }).OpenDirectoryDialog?.('Choose output directory') ?? ''
        if (dir) setOutputDir(dir)
      } catch { /* dialog cancelled */ }
    }

    const handleGenerate = async () => {
      if (!outputDir || selectedIds.size === 0) return
      setGenerating(true)
      setResult(null)
      try {
        const selected = requests.filter((r) => selectedIds.has(r.id))
        const inputJSON = JSON.stringify({ serverName, requests: selected })
        const errMsg = await MCPServerGenerator.Generate(inputJSON, outputDir)
        if (errMsg) {
          setResult({ success: false, message: errMsg })
        } else {
          setResult({ success: true, message: `Server generated in ${outputDir}. Run: cd "${outputDir}" && npm install && npx ts-node index.ts` })
        }
      } catch (e) {
        setResult({ success: false, message: String(e) })
      } finally {
        setGenerating(false)
      }
    }

    return (
      <div className="flex flex-col h-full p-4 gap-4 overflow-y-auto">
        <div>
          <h2 className="text-[12px] font-semibold text-text-1 mb-1">Generate MCP Server</h2>
          <p className="text-[10px] text-text-3">
            Export API endpoints as an MCP Server compatible with Cursor, Claude Code, and other AI assistants.
          </p>
        </div>

        {/* Server name */}
        <div>
          <label className="text-[10px] font-medium text-text-2 block mb-1">Server Name</label>
          <input
            value={serverName}
            onChange={(e) => setServerName(e.target.value)}
            placeholder="my-api-server"
            className="w-full h-7 px-2 text-[10px] font-mono bg-surface-1 border border-border-2 rounded text-text-1 focus:border-accent outline-none"
          />
        </div>

        {/* Collection picker */}
        <div>
          <label className="text-[10px] font-medium text-text-2 block mb-1">Source Collection</label>
          <select
            value={selectedCollectionId}
            onChange={(e) => { setSelectedCollectionId(e.target.value); setSelectedIds(new Set()) }}
            className="w-full h-7 px-2 text-[10px] bg-surface-1 border border-border-2 rounded text-text-1 focus:border-accent outline-none"
          >
            <option value="">— Select a collection —</option>
            {collections.map((c) => (
              <option key={c.id} value={c.id}>{c.name}</option>
            ))}
          </select>
        </div>

        {/* Endpoint checkboxes */}
        {requests.length > 0 && (
          <div>
            <div className="flex items-center justify-between mb-1">
              <label className="text-[10px] font-medium text-text-2">Endpoints to expose ({selectedIds.size}/{requests.length})</label>
              <button onClick={toggleAll} className="text-[9px] text-accent hover:underline">{selectedIds.size === requests.length ? 'Deselect all' : 'Select all'}</button>
            </div>
            <div className="border border-border-1 rounded overflow-hidden divide-y divide-border-0 max-h-[200px] overflow-y-auto">
              {requests.map((r) => (
                <label key={r.id} className="flex items-center gap-2 px-3 py-2 cursor-pointer hover:bg-surface-1 transition-colors">
                  <input
                    type="checkbox"
                    checked={selectedIds.has(r.id)}
                    onChange={() => toggleOne(r.id)}
                    className="w-3 h-3 accent-[var(--color-accent)]"
                  />
                  <span className={cn('text-[9px] font-mono px-1 py-0.5 rounded shrink-0', {
                    'bg-blue-900/30 text-blue-300': r.method === 'GET',
                    'bg-green-900/30 text-green-300': r.method === 'POST',
                    'bg-yellow-900/30 text-yellow-300': r.method === 'PUT' || r.method === 'PATCH',
                    'bg-red-900/30 text-red-300': r.method === 'DELETE',
                    'bg-surface-2 text-text-4': !r.method,
                  })}>
                    {r.method ?? '?'}
                  </span>
                  <span className="text-[10px] text-text-1 truncate flex-1">{r.name}</span>
                </label>
              ))}
            </div>
          </div>
        )}

        {/* Output directory */}
        <div>
          <label className="text-[10px] font-medium text-text-2 block mb-1">Output Directory</label>
          <div className="flex gap-2">
            <input
              value={outputDir}
              onChange={(e) => setOutputDir(e.target.value)}
              placeholder="/path/to/output"
              className="flex-1 h-7 px-2 text-[10px] font-mono bg-surface-1 border border-border-2 rounded text-text-1 focus:border-accent outline-none"
            />
            <button
              onClick={handlePickDir}
              className="h-7 px-3 text-[10px] bg-surface-2 text-text-2 rounded border border-border-2 hover:bg-surface-3 transition-colors flex items-center gap-1.5"
            >
              <FolderOpen size={11} />
              Browse
            </button>
          </div>
        </div>

        {/* Result */}
        {result && (
          <div className={cn('flex items-start gap-2 px-3 py-2 rounded border text-[10px]',
            result.success
              ? 'bg-green-950/30 border-green-900/40 text-green-300'
              : 'bg-red-950/30 border-red-900/40 text-red-300'
          )}>
            {result.success ? <CheckCircle2 size={12} className="shrink-0 mt-0.5" /> : <AlertCircle size={12} className="shrink-0 mt-0.5" />}
            <span>{result.message}</span>
          </div>
        )}

        {/* Generate button */}
        <button
          onClick={handleGenerate}
          disabled={generating || !outputDir || selectedIds.size === 0}
          className="w-full h-9 text-[11px] bg-accent text-white rounded hover:bg-accent/90 disabled:opacity-40 disabled:cursor-not-allowed transition-colors flex items-center justify-center gap-2"
        >
          <Wand2 size={13} />
          {generating ? 'Generating…' : `Generate MCP Server (${selectedIds.size} tools)`}
        </button>
      </div>
    )
  }
  ```
